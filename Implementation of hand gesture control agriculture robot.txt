#include <Wire.h>
#include <MPU6050.h>
#include <SoftwareSerial.h>

MPU6050 mpu;
SoftwareSerial BT(10, 11); // RX, TX

int16_t ax, ay, az;

char command = 'S';

void setup() {
  Wire.begin();
  Serial.begin(9600);
  BT.begin(9600);

  mpu.initialize();

  if (!mpu.testConnection()) {
    Serial.println("MPU6050 connection failed");
    while (1);
  }

  Serial.println("MPU6050 Ready...");
}

void loop() {
  mpu.getAcceleration(&ax, &ay, &az);

  Serial.print("X: "); Serial.print(ax);
  Serial.print(" Y: "); Serial.print(ay);
  Serial.print(" Z: "); Serial.println(az);

  // Gesture Logic
  if (ay > 12000) {
    command = 'F';  // Forward
  }
  else if (ay < -12000) {
    command = 'B';  // Backward
  }
  else if (ax > 12000) {
    command = 'R';  // Right
  }
  else if (ax < -12000) {
    command = 'L';  // Left
  }
  else {
    command = 'S';  // Stop
  }

  BT.write(command);
  Serial.print("Command: ");
  Serial.println(command);

  delay(200);
}
